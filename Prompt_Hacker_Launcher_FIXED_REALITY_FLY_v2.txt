local function fetchSource(url)
    local lastError

    -- Основной способ, обычно поддерживается Roblox executors.
    local ok, result = pcall(function()
        return game:HttpGet(url)
    end)
    if ok and type(result) == "string" and #result > 0 then
        return result
    end
    lastError = tostring(result)

    -- Резервный способ для executors с request API.
    local requestFn =
        (type(request) == "function" and request)
        or (type(http_request) == "function" and http_request)
        or (syn and type(syn.request) == "function" and syn.request)
        or nil

    if requestFn then
        local reqOk, response = pcall(function()
            return requestFn({
                Url = url,
                Method = "GET",
                Headers = { ["Cache-Control"] = "no-cache" }
            })
        end)

        if reqOk and type(response) == "table" then
            local body = response.Body or response.body
            local code = tonumber(response.StatusCode or response.Status or 200)
            if code and code >= 200 and code < 300
                and type(body) == "string" and #body > 0 then
                return body
            end
            lastError = "HTTP " .. tostring(code) .. ": " .. tostring(body)
        elseif not reqOk then
            lastError = tostring(response)
        end
    end

    error("Не удалось скачать файл. " .. tostring(lastError))
end

local busy = false
local function launch(url, label)
    if busy then return end
    busy = true
    status.Text = "Загрузка " .. label .. "..."
    status.TextColor3 = Color3.fromRGB(255, 220, 90)

    task.spawn(function()
        local ok, err = pcall(function()
            if type(loadstring) ~= "function" then
                error("Executor не поддерживает loadstring.")
            end

            local source = fetchSource(url)
            if not source or source == "" then
                error("GitHub вернул пустой файл.")
            end

            local chunk, compileError = loadstring(source)
            if not chunk then
                error("Ошибка компиляции: " .. tostring(compileError))
            end

            status.Text = "Запускаю " .. label .. "..."
            chunk()
            gui:Destroy()
        end)

        if not ok then
            busy = false
            status.Text = "Ошибка: " .. tostring(err)
            status.TextColor3 = Color3.fromRGB(255, 100, 100)
            warn("[Prompt_Hacker Launcher] " .. tostring(err))
        end
    end)
end
